import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Scanner;


public class Main {
    public static String ejecutarComando(String comando) {
        StringBuilder salida = new StringBuilder();


        try {
            ProcessBuilder builder = new ProcessBuilder(comando);
            Process proceso = builder.start();
            BufferedReader reader = new BufferedReader(newInputStreamReader(proceso.getInputStream()));
            String linea;


            while ((linea = reader.readLine()) != null) {
                salida.append(linea).append("\n");
            }
            proceso.waitFor();
        } catch (Exception e) {
            e.printStackTrace();
        }


        return salida.toString();
    }


    // Función para limpiar la pantalla
    public static void limpiarPantalla() {
        try {
            new ProcessBuilder("cmd", "/c", "cls").inheritIO().start().waitFor();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }


    // Función para pausar y esperar a que el usuario pulse una tecla
    public static void esperarTecla() {
        System.out.println("\nPresiona Enter para continuar...");
        try {
            System.in.read();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }


    // Función para mostrar información de la CPU desde la salida del comando systeminfo
    public static void mostrarInfoCPU() {
        limpiarPantalla();
        String resultado = ejecutarComando("systeminfo");
        System.out.println("\n--- Información de la CPU ---");


        // Busca la línea que contiene información relevante del procesador
        for (String linea : resultado.split("\n")) {
            if (linea.contains("Procesador")) {
                System.out.println(linea.trim());
            }
        }
        esperarTecla();
        limpiarPantalla();
    }


    // Función para mostrar información de la memoria instalada desde la salida del comando systeminfo
    public static void mostrarInfoMemoria() {
        limpiarPantalla();
        String resultado = ejecutarComando("systeminfo");
        System.out.println("\n--- Información de la Memoria ---");


        // Busca la línea que contiene información sobre la memoria física instalada
        for (String linea : resultado.split("\n")) {
            if (linea.contains("Memoria")) {
                System.out.println(linea.trim());
            }
        }
        esperarTecla();
        limpiarPantalla();
    }


    // Función para mostrar información del sistema operativo desde la salida del comando systeminfo
    public static void mostrarInfoSistema() {
        limpiarPantalla();
        String resultado = ejecutarComando("systeminfo");
        System.out.println("\n--- Información del Sistema Operativo ---");


        // Busca las líneas que contienen información sobre el nombre y la versión del sistema operativo
        for (String linea : resultado.split("\n")) {
            if (linea.contains("Nombre del sistema operativo:") || linea.contains("Versi�n del sistema operativo:")) {
                System.out.println(linea.trim());
            }
        }
        esperarTecla();
        limpiarPantalla();
    }


    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int opcion;


        do {
            System.out.println("\nMenú de opciones:");
            System.out.println("1. Mostrar la información de la CPU");
            System.out.println("2. Mostrar la información de la memoria instalada");
            System.out.println("3. Nombre y versión del sistema operativo");
            System.out.println("0. Salir");
            System.out.print("Selecciona una opción: ");
            opcion = scanner.nextInt();


            switch (opcion) {
                case 1:
                    mostrarInfoCPU();
                    break;
                case 2:
                    mostrarInfoMemoria();
                    break;
                case 3:
                    mostrarInfoSistema();
                    break;
                case 0:
                    limpiarPantalla();
                    System.out.println("Saliendo del programa...");
                    break;
                default:
                    limpiarPantalla();
                    System.out.println("Opción no válida. Inténtalo nuevamente.");
            }
        } while (opcion != 0);


        scanner.close();
    }
}
